Для уроків 4–7.

2. Цикли

for (i in 0 until n) { }                 // індекси
for (x in arr) { }                       // значення (копія, лише читання)
for (i in arr.indices) { arr[i] = 0 }    // змінюємо елементи
for ((i, x) in arr.withIndex()) { }      // індекс + значення
for (i in arr.lastIndex downTo 0) { }    // назад

while (умова) { }                        // перевірка ПЕРЕД тілом
do { } while (умова)                     // перевірка ПІСЛЯ тіла, ≥ 1 раз
repeat(n) { i -> }                       // n разів (лямбда)
